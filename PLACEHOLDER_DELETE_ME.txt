/**
 * Textos do diagrama.
 *
 * O site inteiro e em ingles, entao o diagrama tambem e: misturar "Capotraste
 * na casa 3" com "Chords used in this tab" na mesma tela ficava incoerente.
 * A versao anterior detectava o idioma do NAVEGADOR - errado, porque o dono
 * navega em portugues e via rotulos traduzidos num site em ingles. Pior: o
 * servidor renderizava em ingles (sem `navigator`) e o cliente trocava para
 * portugues na hidratacao, gerando erro de hidratacao do React.
 */
export function diagramLabels(_locale?: string) {
  return {
    variation: (i: number, n: number) => `Variation ${i} of ${n}`,
    close: "Click anywhere to close",
    startingAt: (f: number) => `Starting at fret ${f}`,
    noDiagram: "No diagram",
    capo: (f: number) => `Capo on fret ${f}`,
    shortened: "Capo",
  };
}
